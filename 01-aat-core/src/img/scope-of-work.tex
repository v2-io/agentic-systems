%  WHAT THIS IS
%  ------------
%  A "map" of the systems AAT covers. Two zones:
%
%    Upper  — the definitional-narrowing SPINE drawn as nested
%             containment frames: each agent class is a proper SUBSET
%             of the one enclosing it (Self-Actuated < Actuated <
%             Agentic < Adaptive). Frame area shrinks inward = the
%             population shrinks to the rare, capability-rich core.
%             Nesting *is* the canonical superset relation (design doc
%             section 2), drawn rather than glyph'd --- it is
%             containment, not a left-to-right pipeline/process.
%    Lower  — ORTHOGONAL property rails that "switch on" (a dot) at
%             the class where the property first becomes meaningful.
%
%  HOW TO EDIT  (this file is built to be refined and extended)
%  -----------
%    * Add / change an agent class:  edit the \scopeclass calls in the
%      "SCOPE DATA" block. One call = one class (outer->inner order).
%      Geometry is derived from the level index, so a new innermost
%      class just needs a new call (check it still fits the canvas).
%    * Add / change a rail:  edit the \rail calls in the "RAIL DATA"
%      block. One call = one rail; its y is derived from the row
%      index, its x-start from the class level it switches on.
%    * Tune geometry/spacing:  the CONFIG block --- every magic number
%      is a named \def with a one-line reason. Nothing is hard-coded
%      twice.
%    * Margins are automatic:  the cream ground is sized to the actual
%      content bounding box + \gM on every side, so edits never
%      reintroduce lopsided margins.
%
%  DESIGN PROVENANCE  (load-bearing "why" --- do not silently revert)
%  -----------------
%    * Containment, not a pipeline. Earlier drafts used arrows; that
%      mis-encoded the relation as a temporal process. Nesting is the
%      canonical proper-subset relation (design doc section 2). Keep
%      it a containment picture.
%    * Continuity-stance rail = the canonical FIVE-value axis from
%      01-aat-core/src/disc-continuity-stance.md (Indifferent /
%      Task-terminal / Instrumentally-continuous / Morally-continuous /
%      Negotiated, in table order). Canon frames it as an *axis* with
%      conceptual (not mathematical) boundaries; the "|" separator
%      reads a touch more discrete than that. That is Joseph's
%      deliberate presentational choice for rail-consistency --- a
%      recorded trade, NOT drift. Do not "restore" a gradient marker
%      thinking it is a regression.
%    * Self-Actuated is treated as established (not hypothetical).
%      This is Joseph's epistemic call, not a result verified in this
%      file; it is why this class is styled like the others rather
%      than reserved/dashed, and why no self-actuation gate is drawn.
%    * Vol-1 scope: Logogenic / Logozoetic agents (03-llm-core /
%      04-eli-core) and the model x objective 2x2 inset are out of
%      scope here by design.
%    Fuller verification/design trail (and per-iteration history, now
%    also in git): msc/scope-of-work-ontology-and-figure-2026-05-17.md
%    Palette sampled from 01-aat-core/AAT-cover.svg.
%
%  Build:  lualatex scope-of-work.tex     (or pdflatex)
% ======================================================================

% -- Scope nesting geometry. Frame for class level i (i = 0 is outer):
%      x : [ \zL + i*\zIL , \zR - i*\zIR ]
%      y : [ \zB + i*\zIB , \zT - i*\zIT ]
%    i.e. an exact arithmetic inset per level. Left/top insets carry
%    the nesting read; the bottom inset is smaller than the top so the
%    interior text stack clears the innermost border.
\def\zL{0.3}    \def\zR{16.8}   % level-0 (outer) left / right
\def\zT{11.0}   \def\zB{5.95}   % level-0 (outer) top  / bottom
\def\zIL{3.4}   \def\zIR{0.25}  % per-level left  / right inset
\def\zIT{0.52}  \def\zIB{0.40}  % per-level top   / bottom inset

% -- Interior text stack, offset down from each frame's top edge.
\def\lpad{0.5}   % left pad inside the frame border (whole stack)
\def\oD{0.22}    % delta      row top below frame top
\def\oN{0.64}    % class-name row top below frame top (gap after delta)
\def\oC{1.06}    % capability row top below frame top
\def\capw{2.7cm} % default capability text width (inner-strip width)

% -- Rails. Row r sits at y = \railTop - r*\railPitch; the rule runs
%    from the switch-on class's left edge to \railR; labels are right-
%    aligned at \rlx; choices ride \rsub above the rule.
\def\railTop{5.0}   \def\railPitch{0.85}
\def\railR{16.4}    % rails' right end
\def\rlx{0.0}       % rail-label right edge (a few ems left of x=\zL)
\def\rsub{0.12}     % choice-text rise above the rule (railsub: SW anchor)
\def\dividerY{5.55} % dotted zone divider between spine and rails

\def\gM{0.45}       % uniform cream margin around the content bbox

% ======================================================================
%  ENGINE  --  one definition each for "a class" and "a rail"
% ======================================================================

\tikzset{
  delta/.style  ={font=\scriptsize\itshape, text=spineDeep,
                  anchor=north west},
  raillab/.style={font=\scriptsize\bfseries, text=railPeri,
                  anchor=east},
  railsub/.style={font=\tiny, text=midgray, anchor=south west,
                  inner sep=1pt},
}

% \scopeclass{level}{tint%}{name}{delta (empty = none)}{capability}{capw}
%   Draws the nested frame (fill+border on the "frames" layer) and the
%   interior [delta]/[name]/[capability] stack (on "main", left-aligned
%   at \lpad inside the border). Level 0 passes an empty delta --- it
%   is the base population, nothing is added to reach it.
\newcommand{\scopeclass}[6]{%
  \pgfmathsetmacro{\fL}{\zL+#1*\zIL}%
  \pgfmathsetmacro{\fR}{\zR-#1*\zIR}%
  \pgfmathsetmacro{\fT}{\zT-#1*\zIT}%
  \pgfmathsetmacro{\fB}{\zB+#1*\zIB}%
  \begin{pgfonlayer}{frames}%
    \fill[spineDeep!#2] (\fL,\fB) rectangle (\fR,\fT);%
    \draw[spineDeep!55, rounded corners=3pt, line width=0.9pt]
      (\fL,\fB) rectangle (\fR,\fT);%
  \end{pgfonlayer}%
  \def\tmpdelta{#4}%
  \ifx\tmpdelta\empty\else
    \node[delta] at (\fL+\lpad,\fT-\oD) {#4};%
  \fi
  \node[font=\footnotesize\bfseries, text=spineDeep, anchor=north west]
    at (\fL+\lpad,\fT-\oN) {#3};%
  \node[align=left, text=ink, font=\scriptsize, text width=#6,
    anchor=north west] at (\fL+\lpad,\fT-\oC) {#5};%
}

% \rail{row}{switch-on class level}{accent color}{label}{choices}
%   Draws the rule (from that class's left edge to \railR), the
%   switch-on dot, the right-aligned label, and the choices above.
\newcommand{\rail}[5]{%
  \pgfmathsetmacro{\ry}{\railTop-#1*\railPitch}%
  \pgfmathsetmacro{\rx}{\zL+#2*\zIL}%
  \draw[#3, line width=1.1pt] (\rx,\ry) -- (\railR,\ry);%
  \fill[#3] (\rx,\ry) circle (2.4pt);%
  \node[raillab, text=#3] at (\rlx,\ry) {#4};%
  \node[railsub] at ($(\rx,\ry)+(0.16,\rsub)$) {#5};%
}
