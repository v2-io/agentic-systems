% common/preamble/environments.tex
%
% Segment environments — the visual register for the 19 segment types
% defined in FORMAT.md. Architecture:
%
%   \segmenthead{<Type Label>}{<Title>}{<status>}{<stage>}
%      ... segment body ...
%   \segmentfoot
%
% The kramdown emitter produces these directly. No 19 distinct LaTeX
% environments — one parameterized macro pair, type label as a string. This
% keeps the AAD type taxonomy and the LaTeX taxonomy decoupled: when FORMAT
% grows a 20th type, the build pipeline maps it to the right label and the
% rendering needs no change here.
%
% Numbering rides kaobook's native section counter (reset per chapter).
% Display is kaobook's default arabic-chapter.arabic-section — "Definition
% 1.4, Result 5.7". The custom \segment counter the build used pre-Phase-1c
% is retired; everything now hangs off \section / \thesection.

% cleveref format for kaobook's native `section` counter — overrides what
% kaorefs sets so \cref{seg:foo} renders as a bare number (matching the
% existing prose convention "see Definition #def-foo" → "see Definition
% I.4", with the type word supplied by the author in source).
%
% IMPORTANT: cleveref's format placeholder for the stored label value is
% `#1`, NOT `\thesection`. Using `\thesection` would expand at cref-TIME
% (current counter value) instead of label-TIME (the value recorded when
% \label was emitted). `#2#1#3` is the canonical "hyperref-open + value +
% hyperref-close" idiom.
%
% Registered inside \AtBeginDocument so we run after kaorefs's own hooks
% and our format wins.
\AtBeginDocument{%
  \crefformat{section}{#2#1#3}%
  \Crefformat{section}{#2#1#3}%
  \crefrangeformat{section}{#3#1#4--#5#2#6}%
  \Crefrangeformat{section}{#3#1#4--#5#2#6}%
  \crefmultiformat{section}{#2#1#3}{, #2#1#3}{, #2#1#3}{, #2#1#3}%
  \Crefmultiformat{section}{#2#1#3}{, #2#1#3}{, #2#1#3}{, #2#1#3}%
  % Appendix segments are chapters; \cref{seg:foo} where seg:foo labels an
  % appendix chapter would otherwise pick up kaorefs's default "Chapter X"
  % rendering and produce awkward prose like "see Derivation Chapter A".
  % Mirror the section convention — bare number — so chapter and section
  % cross-refs read identically in running text.
  \crefformat{chapter}{#2#1#3}%
  \Crefformat{chapter}{#2#1#3}%
  \crefrangeformat{chapter}{#3#1#4--#5#2#6}%
  \Crefrangeformat{chapter}{#3#1#4--#5#2#6}%
  \crefmultiformat{chapter}{#2#1#3}{, #2#1#3}{, #2#1#3}{, #2#1#3}%
  \Crefmultiformat{chapter}{#2#1#3}{, #2#1#3}{, #2#1#3}{, #2#1#3}%
}

% Internal section subheads — Formal Expression / Epistemic Status /
% Discussion / Findings. Each renders as a full-width thin rule across
% the segment band, with the subhead label right-aligned beneath. The
% rule does the visual section break; the right-aligned label names it
% quietly without competing with body text for the reader's eye.
%
% Orphan policy: reserve vertical space for the rule, the label, and a
% few lines of the section that follows; discourage a break in between.
\NewDocumentCommand{\segmentsubhead}{m}{%
  \needspace{5\baselineskip}%
  \par\medskip%
  \noindent\textcolor{rule-gray}{\rule{\segmentrulewidth}{0.3pt}}\par%
  \nobreak\nointerlineskip%
  \vskip-0.4\baselineskip%
  \noindent\makebox[\segmentrulewidth][r]{\sffamily\bfseries #1}\par%
  \nobreak\smallskip%
  \nopagebreak[4]%
}

% Custom override of the kao chapter style. Adds "Chapter" / "Appendix"
% small italic word above the scaled chapter number/letter, drops the
% trailing \autodot. Must be called AFTER \setchapterstyle{kao} since
% kao's \chapterstylekao \renewcommands \chapterformat itself — our
% override has to land after kao's reset.
%
% KOMA-Script's \appendix swaps \@chapapp from \chaptername to
% \appendixname (the localized "Chapter" / "Appendix" pair). Reading
% \@chapapp at chapter-rendering time picks up the current value — so
% appendix chapters render "Appendix" automatically without needing a
% separate appendix-format path. \chaptername alone DOESN'T flip;
% that's why an earlier attempt put "Chapter" everywhere.
%
% Wrapped in \makeatletter / \makeatother because @ isn't a letter at
% body-time when \asfChapterFormat is called; defining the macro under
% letter-catcode captures \@chapapp as a known control sequence so
% expansion later doesn't care what @'s catcode is.
%
% The tabular bottom-alignment places the scaled number row at the
% baseline of the parbox chapterlinesformat wraps us in, so the small
% word sits visually above the big glyph without disturbing the kao
% layout's vertical-rule alignment.
\makeatletter
\NewDocumentCommand{\asfChapterFormat}{}{%
  \renewcommand*{\chapterformat}{%
    \mbox{\begin{tabular}[b]{@{}l@{}}%
      {\sffamily\itshape\fontsize{11pt}{13pt}\selectfont\@chapapp}\\[1pt]%
      {\scalebox{2.85}{\thechapter}}%
    \end{tabular}}%
  }%
}
\makeatother

% Working-notes-internal subhead. Same visual register as \segmentsubhead
% but sized to \linewidth (the workingnotes tcolorbox's content area)
% instead of \segmentrulewidth (body + margin column). Without this
% fallback the segment-level makebox extends past the right edge of the
% workingnotes box, since the box has 12pt left/right padding that
% \segmentrulewidth doesn't account for. Slightly smaller font matches
% the workingnotes body register (which is \small).
\NewDocumentCommand{\workingnotessubhead}{m}{%
  \needspace{3\baselineskip}%
  \par\medskip%
  \noindent\textcolor{rule-gray}{\rule{\linewidth}{0.3pt}}\par%
  \nobreak\nointerlineskip%
  \vskip-0.4\baselineskip%
  \noindent\makebox[\linewidth][r]{\sffamily\bfseries\small #1}\par%
  \nobreak\smallskip%
  \nopagebreak[4]%
}

% Width that spans body + margin gutter + margin column. kaobook's body
% column is \textwidth and the margin column sits to its right with
% \marginparsep gap and \marginparwidth width. We compute the union so
% segment-header rules read as page-spanning visual separators rather than
% body-column decorations.
\newcommand{\segmentrulewidth}{\dimexpr\textwidth+\marginparsep+\marginparwidth\relax}

%   #1 = type label (string, e.g. "Result", "Derivation")
%   #2 = title (the human-readable form of the slug)
%   #3 = status (string, e.g. "exact"; empty string = no badge)
%
% The third argument is always required; the kramdown emitter passes the
% empty string when frontmatter has no `status:` set, which suppresses the
% badge via the IfStrEq below.
%

\colorlet{header-tint}{tint-1}

% Dispatch by type-label (the human-readable form passed to \segmenthead's
% first argument): Definition, Result, Derivation, etc. Order follows the
% palette mapping Joseph chose for the live corpus distribution; the two
% currently-unused FORMAT types (Corollary, Aside) take sensible defaults
% (Corollary→Result family, Aside→Detail family) so they Just Work when
% they land in source.
% Mid-gray used for the bullet between type-label and title in segment
% headers. Light enough to feel like punctuation, dark enough to be
% clearly visible (the rule-gray we previously used was effectively
% invisible against the tinted backgrounds).
\definecolor{bullet-mid}{HTML}{666666}

% Stage glyph — promotion-stage indicator on the far right of the segment
% header strip in review mode. Five filled/empty dots map to the five
% post-`old` promotion stages from FORMAT.md. `missing`/`old` yield no
% glyph (those segments don't reach \segmenthead anyway — they go
% through \missingsegment).
% \CIRCLE (filled) and \Circle (empty) from wasysym are size-paired and
% reliably available — both render at the same baseline size, unlike
% \textbullet/\textcircled{} or the EB-Garamond-missing Unicode ●/○.
\NewDocumentCommand{\stageglyph}{m}{%
  {\color{bullet-mid}\footnotesize\IfStrEqCase{#1}{%
    {draft}           {\CIRCLE\,\Circle\,\Circle\,\Circle\,\Circle}%
    {deps-verified}   {\CIRCLE\,\CIRCLE\,\Circle\,\Circle\,\Circle}%
    {claims-verified} {\CIRCLE\,\CIRCLE\,\CIRCLE\,\Circle\,\Circle}%
    {format-clean}    {\CIRCLE\,\CIRCLE\,\CIRCLE\,\CIRCLE\,\Circle}%
    {candidate}       {\CIRCLE\,\CIRCLE\,\CIRCLE\,\CIRCLE\,\CIRCLE}%
  }[]}%
}

% Title is set large + bold (upright) so it reads with weight comparable
% to inline body-bold cues like "Lemma 1." or "Proof." — competing with
% the prose rather than looking decorative.
%
% Upper rule is 2.5x thicker and ~black, lower rule stays thin and gray —
% the asymmetry pulls the eye to the segment opening and lets the closing
% rule fade to a quiet separator. Between the rules: a tcolorbox holding
% the type-label + status-badge + title line on a soft tinted background.
%
% Orphan/widow policy: \needspace asks for enough vertical space to hold
% the header + a few lines of the first paragraph; if it's not available,
% LaTeX breaks the page first and the segment opens on the next page with
% its body intact. \nopagebreak after the bottom rule discourages a break
% between header and the paragraph that follows.
% Fourth arg is the segment's `stage:` frontmatter value, used in review
% mode for the dot-progression glyph on the far right of the header strip.
% Empty string elides it (public-variant builds, or stage-less segments).
% \segmenthead drives off kaobook's native section counter:
%   \refstepcounter{section} — increments + creates the anchor target for
%     \label{seg:...}, which segment_renderer.rb emits immediately after.
%   \sectionmark — populates the running head with the segment title.
%   \addcontentsline — emits the ToC entry. We compose it manually rather
%     than via \section{...} because kaobook's \section would render its
%     own heading above our tinted box (double header). The contentsline
%     uses \protect\numberline{\thesection} so cleveref and the ToC both
%     see the formatted "I.4"-style number.
\NewDocumentCommand{\segmenthead}{m m m m}{%
  \needspace{10\baselineskip}%
  \refstepcounter{section}%
  \sectionmark{#2}%
  \addcontentsline{toc}{section}{\protect\numberline{\thesection}\textit{#1}: #2}%
  \setheadertint{#1}%
  \par\medskip%
  \noindent\begin{tcolorbox}[
    enhanced,
    width            = \segmentrulewidth,
    colback          = header-tint,
    boxrule          = 0pt,
    arc              = 0pt,
    left             = 8pt, right  = 8pt,
    top              = 6pt, bottom = 6pt,
    before skip      = 0pt, after skip = 0pt,
    % Both rules black; top 2x the bottom so the eye registers the
    % segment-opening edge over the closing edge while both stay clearly
    % visible against the tinted background.
    borderline north = {1.5pt}{0pt}{black},
    borderline south = {0.75pt}{0pt}{black},
  ]
  {\sffamily\bfseries #1~\thesection}%
  \IfStrEq{#3}{}{}{\statusbadge{#3}}%
  \nobreak\quad\textcolor{bullet-mid}{\textbullet}\quad%
  {\large\bfseries #2}.%
  \IfStrEq{#4}{}{}{\hfill\stageglyph{#4}}%
  \end{tcolorbox}\par%
  \nobreak\medskip%
  \nopagebreak[4]%
}

% Segment closer — paragraph spacing only; no visible rule. The next
% segment's opener provides its own top border, so a foot rule would only
% produce a doubled separator when segments sit back-to-back. A chapter
% break or page break is enough of a closer for the trailing-segment case.
\NewDocumentCommand{\segmentfoot}{}{%
  \par\medskip%
}

% Appendix-segment chapter opener.
%
% Per the four-volume hierarchy decision (FORMAT-TODO §The Decided
% Hierarchy): appendix segments render directly as \chapter under an
% \appendix\part{...} group — no intermediate Chapter level inside
% Appendices. An appendix segment IS the chapter.
%
% This routes through kaobook's native \chapter so the chapter counter
% advances, kaobook's chapter style (the big margin glyph + title block)
% renders, and the ToC entry is created automatically. The type label is
% folded into the chapter title so the segment's role stays visible in
% the heading and ToC ("Derivation: Sector Condition Derivation" rather
% than a bare descriptive title that hides whether the segment is a
% Derivation, Discussion, or Worked Example).
%
% Status badge + stage glyph render as a small indicator strip beneath
% the kaobook chapter heading, in the same register the section-level
% \segmenthead uses but without the tinted box (the chapter glyph is
% the visual anchor here; the strip is supplementary). Empty #3 / #4 are
% honored — public-variant builds suppress the stage glyph that way.
%
% Cross-references: segment_renderer.rb emits \label{seg:<slug>} on the
% line after this macro, which captures the chapter counter via the
% standard \chapter → \refstepcounter{chapter} chain.
% Appendix-region ToC downshift: route the auto \addcontentsline call
% that \chapter emits at chapter level through \l@section instead, so
% appendix-chapter ToC entries inherit section-style indent / weight /
% line density and read as a continuation of the segment register
% rather than as a parallel block of bold chapter rows. The body
% chapter heading itself is unaffected (kao margin glyph + title
% render normally).
%
% Implementation: globally redefine \addcontentsline once \appendix
% has fired (via \asfAppendixToCremap below, called by the build
% script). KOMA-Script defers the \chapter-time \addcontentsline call
% past any local group, so the rebind has to be global to take effect.
% Part-level lines are preserved (only chapter→section is redirected).
%
% This wrapper exists once; build-monograph emits \asfAppendixToCremap
% on the same line as \appendix, before the first appendix \part.
% Names without `@` so the macro can be invoked safely from body.tex
% (where `@` is not a letter category code).
\NewDocumentCommand{\asfAppendixToCremap}{}{%
  \let\asfOrigAddCL\addcontentsline%
  \renewcommand{\addcontentsline}[3]{%
    \def\asfTmpa{##2}\def\asfTmpb{chapter}%
    \ifx\asfTmpa\asfTmpb%
      \asfOrigAddCL{##1}{section}{##3}%
    \else%
      \asfOrigAddCL{##1}{##2}{##3}%
    \fi%
  }%
}

\NewDocumentCommand{\segmentappendixchapter}{m m m m}{%
  % Body heading is a full kaobook chapter (margin glyph + title); the
  % auto-emitted ToC line is rerouted to section level by the global
  % \asfAppendixToCremap rebind that fires once at \appendix time.
  \chapter[\textit{#1}: #2]{\textit{#1}: #2}%
  % Indicator strip — italic type label is redundant with the chapter
  % title but provides anchoring; status + stage carry the new info.
  \IfStrEq{#3}{}{%
    \IfStrEq{#4}{}{}{%
      \par\noindent\hfill\stageglyph{#4}\par\medskip%
    }%
  }{%
    \par\noindent%
    \statusbadge{#3}%
    \IfStrEq{#4}{}{}{\hfill\stageglyph{#4}}%
    \par\medskip%
  }%
}

% Missing-segment placeholder — for slugs listed in OUTLINE.md but without
% a corresponding src/ file. Generates a TOC entry and a visible marker so
% reviewers see the gap in the rendered output, not just in a sidecar list.
% Rules are the same full-width treatment as \segmenthead so the visual
% break is consistent for landed and missing segments.
\NewDocumentCommand{\missingsegment}{m m m}{%
  % #1 = type label, #2 = slug, #3 = claim text (from the outline table)
  % ToC entry uses the slug rather than the claim text (claims are often
  % paragraph-length and would render unreadably as a ToC line); the
  % full claim appears in the visible marker below.
  \needspace{8\baselineskip}%
  \refstepcounter{section}%
  \sectionmark{\##2}%
  \addcontentsline{toc}{section}{\protect\numberline{\thesection}\textit{#1}: \texttt{\##2} \textnormal{\textit{(missing)}}}%
  % Emit the seg: label even though the segment isn't written — cross-refs
  % from other segments still resolve to a number rather than ??.
  \label{seg:#2}%
  \par\medskip%
  \noindent{\color{status-tentative}\rule{\segmentrulewidth}{0.75pt}}\par%
  \nobreak\smallskip%
  \noindent%
  {\sffamily\bfseries\color{status-tentative} #1~\thesection}%
  \quad\textcolor{bullet-mid}{\textbullet}\quad%
  {\large\bfseries\color{status-tentative} \texttt{\##2}}.%
  \par\nobreak\smallskip%
  \noindent\emph{Segment \texttt{#2} not yet written.}\par%
  \par\smallskip%
  \IfStrEq{#3}{}{}{\noindent\textit{Claim:} #3\par\nobreak\smallskip}%
  \noindent{\color{status-tentative}\rule{\segmentrulewidth}{0.75pt}}\par%
  \nobreak\medskip%
}

% Missing appendix-chapter placeholder — appendix segments listed in
% OUTLINE.md without a corresponding src/ file. Mirrors \missingsegment
% but for the chapter level (appendix segments are chapters), so the
% missing placeholder advances the chapter counter rather than orphaning
% itself under whatever section last fired. Renders a \chapter with the
% claim text as title and a marker block underneath.
\NewDocumentCommand{\missingappendixchapter}{m m m}{%
  % #1 = type label, #2 = slug, #3 = claim text (from the outline table)
  % Chapter title uses the slug (stable, short identifier) rather than the
  % claim text — claim columns in OUTLINE.md are often paragraph-length
  % and would spill into the ToC unreadably. The full claim renders in
  % the body marker below the chapter heading.
  \chapter[\textit{#1}: \texttt{\##2} \textnormal{\textit{(missing)}}]{\textit{#1}: \texttt{\##2} \textnormal{\textit{(missing)}}}%
  \label{seg:#2}%
  \par\medskip%
  \noindent{\color{status-tentative}\rule{\segmentrulewidth}{0.75pt}}\par%
  \nobreak\smallskip%
  \noindent\emph{Segment \texttt{#2} not yet written.}\par%
  \par\smallskip%
  \noindent\textit{Claim:} #3\par%
  \nobreak\smallskip%
  \noindent{\color{status-tentative}\rule{\segmentrulewidth}{0.75pt}}\par%
  \nobreak\medskip%
}

% Segment epigraph — summary block between the segment title and the
% first \segmentsubhead ("Formal Expression"). Slightly smaller, justified,
% centered horizontally on the full segment band (\segmentrulewidth).
%
% Implementation: capture the body into an lrbox, then place it inside a
% \makebox of width \segmentrulewidth with center-alignment. The \makebox
% is rendered \noindent so it begins flush with the body column's left
% edge; its visual extent then runs from x=0 to x=\segmentrulewidth, with
% the inner parbox (width 0.7 * segmentrulewidth) horizontally centered.
%
% Why not adjustwidth*: on chapter-first pages the kao chapter style
% transitions through \pagelayout{wide}, leaving the page in a state
% where adjustwidth* samples wider dimensions than the surrounding body
% prose actually uses, and the epigraph then extends past the page's
% left edge. \makebox doesn't sample those page-state dimensions; the
% positioning is fully determined by the explicit width we hand it.
\newsavebox{\segepibox}

\NewDocumentEnvironment{segmentepigraph}{}{%
  % \medskip leading (was \smallskip) so the gap above the epigraph
  % matches the gap above the header and the gap below the epigraph
  % (header→epigraph and epigraph→first-subhead now both read as
  % section-level breaks rather than tightly-stacked content).
  \par\medskip%
  \begin{lrbox}{\segepibox}%
  \begin{minipage}{0.7\segmentrulewidth}%
  \small%
  % A bit of parskip so multi-paragraph epigraphs breathe — minipage
  % defaults to \parskip=0 which is too tight when the summary runs to
  % two or three paragraphs.
  \setlength{\parskip}{0.8\baselineskip}%
  \setlength{\parindent}{0pt}%
  \noindent\ignorespaces%
}{%
  \par%
  \end{minipage}%
  \end{lrbox}%
  \noindent\makebox[\segmentrulewidth][c]{\usebox{\segepibox}}%
  \par\medskip%
}

% Wide-section wrapper — for Discussion and Findings, where the section
% content extends to the full segment band (body + margin gutter +
% margin column). No background, no rule; just a wider text area.
% tcolorbox's `blanker` skin gives us width control without decoration.
%
% Parskip is bumped inside the wrapper so multi-paragraph commentary in
% Discussion / Findings (which is interpretive prose, often several
% paragraphs of explanation) reads with proper breathing room between
% paragraphs rather than the body-default tight stacking.
\NewDocumentEnvironment{segmentwidesection}{}{%
  \par\medskip%
  \begin{tcolorbox}[
    blanker, breakable,
    width        = \segmentrulewidth,
    left         = 0pt, right  = 0pt,
    top          = 0pt, bottom = 0pt,
    before skip  = 0pt, after skip = 0pt,
  ]%
  \setlength{\parskip}{0.8\baselineskip}%
  \setlength{\parindent}{0pt}%
}{%
  \end{tcolorbox}%
  \par\medskip%
}

% Callouts — Obsidian-style `> [!type] title` blocks. We tint by type

    borderline west = {2.5pt}{0pt}{callout-rule},
  ]%
  {\sffamily\bfseries\color{callout-rule}\MakeUppercase{#1}%

% Working Notes — review-mode-only environment for the trailing ## Working
% Notes section of each segment. Background-tinted box spanning body +
% margin gutter + margin column (so the notes visibly step out of the main
% column register), shrunk and dimmed so the eye reads them as process
% chatter rather than load-bearing prose. The renderer only opens the env
% if the segment actually has a Working Notes heading, so no empty boxes.
\usepackage{tcolorbox}
\tcbuselibrary{breakable}

\definecolor{wn-bg}{HTML}{FAF7E5}  % subtle cream-tinted off-white
\definecolor{wn-fg}{HTML}{6E6354}  % warm muted gray-brown

% colupper / fontupper carry the text styling as persistent properties of
% the tcolorbox rather than as inline `\color` / `\small` directives —
% which is what lets them survive a page break inside a breakable box.
% (The inline form resets to black when the second tcolorbox fragment
% begins on the next page, a recurring KOMA + tcolorbox surprise.)
\NewDocumentEnvironment{workingnotes}{}{%
  \par\bigskip%
  \begin{tcolorbox}[
    enhanced, breakable,
    width        = \segmentrulewidth,
    colback      = wn-bg,
    colframe     = wn-bg,
    colupper     = wn-fg,
    fontupper    = \small,
    boxrule      = 0pt,
    arc          = 0pt,
    left         = 12pt, right  = 12pt,
    top          = 10pt, bottom = 10pt,
    before skip  = 0pt,  after skip = 0pt,
  ]%
  {\sffamily\bfseries Working Notes.}\par\smallskip%
}{%
  \end{tcolorbox}%
  \par\medskip%
}
