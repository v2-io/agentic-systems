\documentclass[tikz,border=6pt]{standalone}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily,
  >=Latex,
  agent/.style={draw, rounded corners=3pt, fill=blue!8, align=center, minimum width=2.7cm, minimum height=0.82cm},
  box/.style={draw, rounded corners=3pt, fill=green!10, align=center, minimum width=3.2cm, minimum height=0.82cm},
  warn/.style={draw, rounded corners=3pt, fill=orange!12, align=center, font=\sffamily\scriptsize}
]

\node[agent] (A) {agent $A$\\future action};
\node[agent, right=4.2cm of A] (B) {observer $B$\\filtration $\mathcal F_B^t$};
\node[box, above=0.9cm of $(A)!0.5!(B)$] (hb)
  {$H_b^{A\mid B}(t,\tau)$\\action unpredictability};

\draw[->, thick] (A.east) -- node[below, font=\sffamily\scriptsize] {prediction channel} (B.west);
\draw[->, thick] (hb.south) -- ($(A.east)!0.5!(B.west)$);

\node[box, below=1.0cm of A, fill=yellow!14] (coop)
  {cooperative value\\low $H_b^{A\mid B}$ helps coordination};
\node[box, below=1.0cm of B, fill=red!7] (adv)
  {adversarial value\\high $H_b^{A\mid B}$ helps surprise};

\draw[->, thick, green!55!black] (hb.west) .. controls +(-1.0,-0.7) and +(0,0.9) .. (coop.north);
\draw[->, thick, red!70!black] (hb.east) .. controls +(1.0,-0.7) and +(0,0.9) .. (adv.north);

\node[warn, above=0.55cm of hb, xshift=-2.35cm, minimum width=3.55cm] (entropy)
  {entropy needs discrete/measure\\convention and finite $H_b^{\max}$};
\draw[->, dashed, orange!85!black] (entropy.south) -- (hb.north);

\node[warn, below=0.75cm of $(coop)!0.5!(adv)$, minimum width=5.2cm] (direction)
  {keep directions separate: attacker opacity $H_b^{A|B}$ vs target opacity $H_b^{B|A}$};
\draw[->, dashed, orange!85!black] (direction.north west) -- (coop.south);
\draw[->, dashed, orange!85!black] (direction.north east) -- (adv.south);

\node[warn, above=0.55cm of hb, xshift=2.65cm, minimum width=3.85cm] (mult)
  {opacity should modulate $\gamma$, not erase\\predictable disturbance by default};
\draw[->, dashed, orange!85!black] (mult.south) -- (hb.north east);

\end{tikzpicture}
