% Agentic Systems Framework — monograph build
%
% Toolchain: LuaLaTeX + biber. Entrypoint compiled by mono/build.
% Class: kaobook (vendored at vendor/kaobook/; TEXINPUTS set by the build script).
%
% v0.1.0 minimal scaffold: structure + fonts + math packages, no content yet
% beyond a Hello-AAT page proving the pipeline compiles.

\documentclass[
  fontsize    = 10pt,
  twoside     = false,     % single-sided during iteration; predictable right margin
  open        = any,
  fontmethod  = modern,    % unicode-math; lets us override with fontspec
  secnumdepth = 3,         % Part / Chapter / Section (= Segment) all numbered
]{kaobook}

\input{preamble/setup}

% Kaobook feature modules
\usepackage{kaotheorems}   % theorem environments (amsthm-based)
% mathtools and kaorefs both define \refeq; let kaorefs win since we use
% its \cref-style sibling commands consistently.
\let\refeq\relax
\usepackage{kaorefs}       % cleveref + hyperref wrappers

% Project-specific rendering machinery — status badges first (referenced by
% segment headers), then environments, then equation-tag margin commands.
\input{preamble/status-badges}
\input{preamble/environments}
\input{preamble/eq-tags}
\usepackage[natbib=true]{kaobiblio}
\addbibresource{references.bib}

%----------------------------------------------------------------------------
% Document metadata
%----------------------------------------------------------------------------
\titlehead{}
% Build stamp — generated by bin/build-monograph into the stage. Exposes
% \buildsemver, \buildsha, \builddate macros for the title page footer.
\input{build-info}
% Generated alongside references.bib. Defines \ifhasreferences so empty
% current volumes do not render an empty bibliography heading.
\input{references-info}

\title{The Agentic Systems Framework}
\subtitle{Adaptation and Actuation Theory, with Temporal Software Theory,
          Logogenic Agents, and Emergent Logozoetic Intelligences}
\author{Joseph A. Wecker}
\date{\today}
\publishers{v\buildsemver\ snapshot \textbullet\ \buildsha\ \textbullet\ \builddate}

% Show segments (= sections) in the ToC. Subsection / subsubsection are
% segment-internal fields ("Formal Expression", named areas like "Search
% Log") and stay out of the ToC by default; bump tocdepth if a future
% need surfaces.
\setcounter{tocdepth}{3}

\begin{document}

\frontmatter
% Cover-page emission. The cover SVG (declared in mono-meta.yaml) is
% pre-rendered to PDF by the build script (rsvg-convert) and exposed
% here as \volumecoverpath. An empty path means "no cover artwork
% configured" (or rsvg-convert failed); skip silently in that case.
\ifdefempty{\volumecoverpath}{}{%
  \includepdf[pages=1,fitpaper=true]{\volumecoverpath}%
}

% Table of contents. \ifvolumetoc is set by the build script from
% mono-meta.yaml `toc:` (default true) and overridden by `--no-toc`.
\ifvolumetoc
  \tableofcontents
\fi

% body.tex is the typeset rendering of the per-volume assembled markdown
% (mono/<slug>-v<sem>.md), produced by Mono::Typeset / AsfVolumeLatex.
% It begins in \frontmatter scope (volume preface via \addchap), emits
% \mainmatter + \setchapterstyle{kao} on the first \part / \appendices,
% and runs through to the last appendix segment. See
% msc/markdown-first-pipeline.md for the pipeline architecture.
\input{body}

\ifhasreferences
  \printbibliography[heading=bibintoc,title={Bibliography}]
\fi

\end{document}
