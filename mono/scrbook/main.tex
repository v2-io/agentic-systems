% Agentic Systems Framework — scrbook (vanilla KOMA-Script) build target.
%
% Distinct from common/main.tex (kaobook target). This entrypoint produces a
% classical math-monograph rendering of the same per-volume assembled
% markdown (mono/<slug>-v<sem>.md): single column, no Tufte margin notes, no
% custom chapter glyphs, no per-segment tinted boxes. The visual register
% aspires to a clean technical book — Lang, Folland, the AMS GSM series.
%
% Toolchain: LuaLaTeX (for fontspec / unicode math), KOMA-Script scrbook,
% amsmath family. No vendored class — KOMA is in TeXLive.

\documentclass[
  fontsize         = 10pt,
  paper            = letter,
  twoside          = true,
  open             = right,    % chapters open on right (recto) — twoside convention
  DIV              = 12,       % wider body column (KOMA default ≈ 10–11)
  BCOR             = 10mm,     % binding correction reserved on inner edge
  numbers          = noenddot,
  toc              = listof,
  toc              = bibliography,
  bibliography     = totoc,
  parskip          = half-,    % half-line paragraph gap; no first-line indent
  headings         = standardclasses,
]{scrbook}

\input{preamble/setup}
\usepackage[
  backend=biber,
  citestyle=numeric-comp,
  bibstyle=numeric,
  sorting=none,
  natbib=true,
  hyperref=true,
  backref=true,
]{biblatex}
\addbibresource{references.bib}
\input{preamble/segment}
\input{preamble/environments}

%----------------------------------------------------------------------------
% Per-build metadata. Generated by bin/build-scrbook into the build stage —
% defines \buildsemver, \buildsha, \builddate, \volumetitle, \volumeshorttitle,
% \volumeslug, \volumecoverpath, and \ifvolumetoc.
%----------------------------------------------------------------------------
\input{build-info}
% Generated alongside references.bib. Defines \ifhasreferences so empty
% current volumes do not render an empty bibliography heading.
\input{references-info}

\title{\volumetitle}
\author{Joseph A. Wecker}
\date{\today}
\publishers{v\buildsemver\ \textbullet\ \buildsha\ \textbullet\ \builddate}

% Number Part / Chapter / Section in the body and ToC. Subsections (segment
% fields like Formal Expression) and below stay unnumbered — they're
% structural, not citable.
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{2}

\begin{document}

\frontmatter

% Cover artwork. The build script renders mono-meta.yaml's cover_svg via
% rsvg-convert into the stage and exposes the relative path as
% \volumecoverpath. Empty path → no cover configured; skip silently.
\ifdefempty{\volumecoverpath}{}{%
  \includepdf[pages=1,fitpaper=true]{\volumecoverpath}%
}

\maketitle

\ifvolumetoc
  \tableofcontents
\fi

% body.tex is the typeset rendering of the per-volume assembled markdown
% (mono/<slug>-v<sem>.md), produced by Mono::TypesetScrbook /
% AsfScrbookLatex. It begins in \frontmatter scope (volume preface via
% \addchap), emits \mainmatter on the first \part / appendices group, and
% runs through to the last appendix segment.
\input{body}

\ifhasreferences
  \printbibliography[heading=bibintoc,title={Bibliography}]
\fi

\end{document}
